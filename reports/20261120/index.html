WITH controls AS (SELECT toDate(toTimeZone(timestamp,'Asia/Shanghai')) AS day,person_id,toString(properties.device_model) AS model,if(event='screen_view','control','start') AS action FROM events WHERE person_mode='full' AND timestamp>=toDateTime('2026-10-21 00:00:00','Asia/Shanghai') AND timestamp<toDateTime('2026-11-20 00:00:00','Asia/Shanghai') AND properties.screen_title='pump_control_page' AND (event='screen_view' OR (event='app_element_click' AND properties.element_name IN ('start_pumping','start_pumping_button')))), home_rows AS (SELECT toDate(toTimeZone(timestamp,'Asia/Shanghai')) AS day,person_id,if(event='screen_view','home','start') AS action,arrayJoin(JSONExtractArrayRaw(toString(properties.extra_data),'device')) AS device FROM events WHERE person_mode='full' AND timestamp>=toDateTime('2026-10-21 00:00:00','Asia/Shanghai') AND timestamp<toDateTime('2026-11-20 00:00:00','Asia/Shanghai') AND properties.screen_title='device_sub_home' AND properties.device_type='breast_pump' AND (event='screen_view' OR (event='app_element_click' AND properties.module_name='device_control_card' AND properties.element_name='play_button'))), actions AS (SELECT day,person_id,model,action FROM controls UNION ALL SELECT day,person_id,JSONExtractString(device,'device_model') AS model,action FROM home_rows WHERE JSONExtractBool(device,'is_selected_device') OR JSONExtractString(device,'is_selected_device')='true') SELECT day,model,uniqExactIf(person_id,action='home') AS home_users,uniqExactIf(person_id,action='control') AS control_users,uniqExactIf(person_id,action='start') AS start_users FROM actions GROUP BY day,model ORDER BY day,model
